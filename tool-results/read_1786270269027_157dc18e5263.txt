    87→    87→function formatDate(iso: string): string {
    88→    88→  try {
    89→    89→    return format(new Date(iso), "d MMM yyyy, HH:mm", { locale: idLocale });
    90→    90→  } catch {
    91→    91→    return iso;
    92→    92→  }
    93→    93→}

    95→    95→function relativeDate(iso: string): string {
    96→    96→  const diff = Date.now() - new Date(iso).getTime();
    97→    97→  const min = Math.floor(diff / 60000);
    98→    98→  if (min < 1) return "baru saja";
    99→    99→  if (min < 60) return `${min} mnt lalu`;
   100→   100→  const hr = Math.floor(min / 60);
   101→   101→  if (hr < 24) return `${hr} jam lalu`;
   102→   102→  const day = Math.floor(hr / 24);
   103→   103→  if (day < 30) return `${day} hari lalu`;
   104→   104→  return formatDate(iso);
   105→   105→}

   107→   107→/** Lead is considered "stale" if it has been NEW or CONTACTED for >48h
   108→   108→ *  since its last update. Used by the dashboard warning card + the
   109→   109→ *  `?stale=1` deep-link filter on the leads page. */
   110→   110→const STALE_MS = 48 * 60 * 60 * 1000;
   111→   111→const STALE_STATUSES = new Set(["NEW", "CONTACTED"]);
   112→   112→
   113→   113→export function isStaleLead(lead: { status: string; updatedAt: string }): boolean {
   114→   114→  if (!STALE_STATUSES.has(lead.status)) return false;
   115→   115→  return Date.now() - new Date(lead.updatedAt).getTime() > STALE_MS;
   116→   116→}

   118→   118→/**
   119→   119→ * Trigger a CSV download of the given leads.
   120→   120→ *
   121→   121→ * If a date range is applied (from/to), the filename follows the pattern
   122→   122→ * `leads-{from}-to-{to}.csv` so the exported file is self-describing.
   123→   123→ * Missing ends use `mulai` (start) / `kini` (now) so the filename stays
   124→   124→ * readable. Without a date range the legacy `pintu-legal-leads-{today}.csv`
   125→   125→ * name is used.
   126→   126→ */
   127→   127→function exportCsv(rows: LeadRow[], from?: string, to?: string) {
   128→   128→  const headers = [
   129→   129→    "Nama",
   130→   130→    "Telepon",
   131→   131→    "Email",
   132→   132→    "Bisnis",
   133→   133→    "Layanan",
   134→   134→    "Status",
   135→   135→    "Sumber",
   136→   136→    "Catatan",
   137→   137→    "Pesan",
   138→   138→    "Dibuat",
   139→   139→  ];
   140→   140→  const lines = [headers.join(",")];
   141→   141→  for (const r of rows) {
   142→   142→    const cells = [
   143→   143→      r.name,
   144→   144→      r.phone,
   145→   145→      r.email,
   146→   146→      r.businessName,
   147→   147→      r.serviceName ?? "",
   148→   148→      r.status,
   149→   149→      r.source,
   150→   150→      r.note,
   151→   151→      r.message,
   152→   152→      r.createdAt,
   153→   153→    ].map((c) => `"${String(c).replace(/"/g, '""')}"`);
   154→   154→    lines.push(cells.join(","));
   155→   155→  }
   156→   156→  const blob = new Blob([lines.join("\n")], { type: "text/csv;charset=utf-8" });
   157→   157→  const url = URL.createObjectURL(blob);
   158→   158→  const a = document.createElement("a");
   159→   159→  a.href = url;
   160→   160→  let filename: string;
   161→   161→  if (from || to) {
   162→   162→    const fromPart = from || "mulai";
   163→   163→    const toPart = to || "kini";
   164→   164→    filename = `leads-${fromPart}-to-${toPart}.csv`;
   165→   165→  } else {
   166→   166→    filename = `pintu-legal-leads-${format(new Date(), "yyyy-MM-dd")}.csv`;
   167→   167→  }
   168→   168→  a.download = filename;
   169→   169→  document.body.appendChild(a);
   170→   170→  a.click();
   171→   171→  document.body.removeChild(a);
   172→   172→  URL.revokeObjectURL(url);
   173→   173→}
