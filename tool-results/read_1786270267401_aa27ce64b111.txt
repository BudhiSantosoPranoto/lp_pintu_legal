    87→function formatDate(iso: string): string {
    88→  try {
    89→    return format(new Date(iso), "d MMM yyyy, HH:mm", { locale: idLocale });
    90→  } catch {
    91→    return iso;
    92→  }
    93→}

    95→function relativeDate(iso: string): string {
    96→  const diff = Date.now() - new Date(iso).getTime();
    97→  const min = Math.floor(diff / 60000);
    98→  if (min < 1) return "baru saja";
    99→  if (min < 60) return `${min} mnt lalu`;
   100→  const hr = Math.floor(min / 60);
   101→  if (hr < 24) return `${hr} jam lalu`;
   102→  const day = Math.floor(hr / 24);
   103→  if (day < 30) return `${day} hari lalu`;
   104→  return formatDate(iso);
   105→}

   107→/** Lead is considered "stale" if it has been NEW or CONTACTED for >48h
   108→ *  since its last update. Used by the dashboard warning card + the
   109→ *  `?stale=1` deep-link filter on the leads page. */
   110→const STALE_MS = 48 * 60 * 60 * 1000;
   111→const STALE_STATUSES = new Set(["NEW", "CONTACTED"]);
   112→
   113→export function isStaleLead(lead: { status: string; updatedAt: string }): boolean {
   114→  if (!STALE_STATUSES.has(lead.status)) return false;
   115→  return Date.now() - new Date(lead.updatedAt).getTime() > STALE_MS;
   116→}

   118→/**
   119→ * Trigger a CSV download of the given leads.
   120→ *
   121→ * If a date range is applied (from/to), the filename follows the pattern
   122→ * `leads-{from}-to-{to}.csv` so the exported file is self-describing.
   123→ * Missing ends use `mulai` (start) / `kini` (now) so the filename stays
   124→ * readable. Without a date range the legacy `pintu-legal-leads-{today}.csv`
   125→ * name is used.
   126→ */
   127→function exportCsv(rows: LeadRow[], from?: string, to?: string) {
   128→  const headers = [
   129→    "Nama",
   130→    "Telepon",
   131→    "Email",
   132→    "Bisnis",
   133→    "Layanan",
   134→    "Status",
   135→    "Sumber",
   136→    "Catatan",
   137→    "Pesan",
   138→    "Dibuat",
   139→  ];
   140→  const lines = [headers.join(",")];
   141→  for (const r of rows) {
   142→    const cells = [
   143→      r.name,
   144→      r.phone,
   145→      r.email,
   146→      r.businessName,
   147→      r.serviceName ?? "",
   148→      r.status,
   149→      r.source,
   150→      r.note,
   151→      r.message,
   152→      r.createdAt,
   153→    ].map((c) => `"${String(c).replace(/"/g, '""')}"`);
   154→    lines.push(cells.join(","));
   155→  }
   156→  const blob = new Blob([lines.join("\n")], { type: "text/csv;charset=utf-8" });
   157→  const url = URL.createObjectURL(blob);
   158→  const a = document.createElement("a");
   159→  a.href = url;
   160→  let filename: string;
   161→  if (from || to) {
   162→    const fromPart = from || "mulai";
   163→    const toPart = to || "kini";
   164→    filename = `leads-${fromPart}-to-${toPart}.csv`;
   165→  } else {
   166→    filename = `pintu-legal-leads-${format(new Date(), "yyyy-MM-dd")}.csv`;
   167→  }
   168→  a.download = filename;
   169→  document.body.appendChild(a);
   170→  a.click();
   171→  document.body.removeChild(a);
   172→  URL.revokeObjectURL(url);
   173→}

   541→function LeadDetailSheet({
   542→  lead,
   543→  open,
   544→  onOpenChange,
   545→  services,
   546→}: {
   547→  lead: LeadRow | null;
   548→  open: boolean;
   549→  onOpenChange: (open: boolean) => void;
   550→  services: ServiceOption[];
   551→}) {
   552→  const [status, setStatus] = React.useState<string>(lead?.status ?? "NEW");
   553→  const [note, setNote] = React.useState<string>(lead?.note ?? "");
   554→  const [saving, setSaving] = React.useState(false);
   555→  const [deleting, setDeleting] = React.useState(false);
   556→
   557→  // Sync local state when lead changes
   558→  React.useEffect(() => {
   559→    if (lead) {
   560→      setStatus(lead.status);
   561→      setNote(lead.note);
   562→    }
   563→  }, [lead?.id]);
   564→
   565→  if (!lead) {
   566→    return (
   567→      <Sheet open={open} onOpenChange={onOpenChange}>
   568→        <SheetContent side="right" className="w-full sm:max-w-md" />
   569→      </Sheet>
   570→    );
   571→  }
   572→
   573→  async function handleSave() {
   574→    if (!lead) return;
   575→    setSaving(true);
   576→    try {
   577→      const res = await fetch(`/api/admin/leads/${lead.id}`, {
   578→        method: "PATCH",
   579→        headers: { "Content-Type": "application/json" },
   580→        body: JSON.stringify({ status, note }),
   581→      });
   582→      const data = await res.json().catch(() => ({}));
   583→      if (!res.ok || !data.ok) {
   584→        toast.error(data.error ?? "Gagal menyimpan perubahan.");
   585→        return;
   586→      }
   587→      toast.success("Lead diperbarui.");
   588→      // Refresh to reflect updates on the table
   589→      window.location.reload();
   590→    } catch {
   591→      toast.error("Kesalahan jaringan saat menyimpan.");
   592→    } finally {
   593→      setSaving(false);
   594→    }
   595→  }
   596→
   597→  async function handleDelete() {
   598→    if (!lead) return;
   599→    if (!confirm("Hapus lead ini secara permanen? Tindakan ini tidak bisa dibatalkan.")) {
   600→      return;
