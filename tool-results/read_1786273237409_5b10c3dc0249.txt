    91→function formatDate(iso: string): string {
    92→  try {
    93→    return format(new Date(iso), "d MMM yyyy, HH:mm", { locale: idLocale });
    94→  } catch {
    95→    return iso;
    96→  }
    97→}
    98→
    99→function relativeDate(iso: string): string {
   100→  const diff = Date.now() - new Date(iso).getTime();
   101→  const min = Math.floor(diff / 60000);
   102→  if (min < 1) return "baru saja";
   103→  if (min < 60) return `${min} mnt lalu`;
   104→  const hr = Math.floor(min / 60);
   105→  if (hr < 24) return `${hr} jam lalu`;
   106→  const day = Math.floor(hr / 24);
   107→  if (day < 30) return `${day} hari lalu`;
   108→  return formatDate(iso);
   109→}

   111→/** Lead is considered "stale" if it has been NEW or CONTACTED for >48h
   112→ *  since its last update. Used by the dashboard warning card + the
   113→ *  `?stale=1` deep-link filter on the leads page. */
   114→const STALE_MS = 48 * 60 * 60 * 1000;
   115→const STALE_STATUSES = new Set(["NEW", "CONTACTED"]);
   116→
   117→export function isStaleLead(lead: { status: string; updatedAt: string }): boolean {
   118→  if (!STALE_STATUSES.has(lead.status)) return false;
   119→  return Date.now() - new Date(lead.updatedAt).getTime() > STALE_MS;
   120→}

   122→/**
   123→ * Trigger a CSV download of the given leads.
   124→ *
   125→ * If a date range is applied (from/to), the filename follows the pattern
   126→ * `leads-{from}-to-{to}.csv` so the exported file is self-describing.
   127→ * Missing ends use `mulai` (start) / `kini` (now) so the filename stays
   128→ * readable. Without a date range the legacy `pintu-legal-leads-{today}.csv`
   129→ * name is used.
   130→ */
   131→function exportCsv(rows: LeadRow[], from?: string, to?: string) {
   132→  const headers = [
   133→    "Nama",
   134→    "Telepon",
   135→    "Email",
   136→    "Bisnis",
   137→    "Layanan",
   138→    "Status",
   139→    "Sumber",
   140→    "Catatan",
   141→    "Pesan",
   142→    "Dibuat",
   143→  ];
   144→  const lines = [headers.join(",")];
   145→  for (const r of rows) {
   146→    const cells = [
   147→      r.name,
   148→      r.phone,
   149→      r.email,
   150→      r.businessName,
   151→      r.serviceName ?? "",
   152→      r.status,
   153→      r.source,
   154→      r.note,
   155→      r.message,
   156→      r.createdAt,
   157→    ].map((c) => `"${String(c).replace(/"/g, '""')}"`);
   158→    lines.push(cells.join(","));
   159→  }
   160→  const blob = new Blob([lines.join("\n")], { type: "text/csv;charset=utf-8" });
   161→  const url = URL.createObjectURL(blob);
   162→  const a = document.createElement("a");
   163→  a.href = url;
   164→  let filename: string;
   165→  if (from || to) {
   166→    const fromPart = from || "mulai";
   167→    const toPart = to || "kini";
   168→    filename = `leads-${fromPart}-to-${toPart}.csv`;
   169→  } else {
   170→    filename = `pintu-legal-leads-${format(new Date(), "yyyy-MM-dd")}.csv`;
   171→  }
   172→  a.download = filename;
   173→  document.body.appendChild(a);
   174→  a.click();
   175→  document.body.removeChild(a);
   176→  URL.revokeObjectURL(url);
   177→}

   198→  const [selectedIds, setSelectedIds] = React.useState<Set<string>>(
   199→    () => new Set()
   200→  );
   201→
   202→  // Date range filter state. `dateFrom`/`dateTo` track the input boxes
   203→  // (uncommitted); `appliedFrom`/`appliedTo` are the values actually used
   204→  // by the filter. The "Filter" button commits the inputs; "Reset" clears
   205→  // both layers.
   206→  const [dateFrom, setDateFrom] = React.useState("");
   207→  const [dateTo, setDateTo] = React.useState("");
   208→  const [appliedFrom, setAppliedFrom] = React.useState("");
   209→  const [appliedTo, setAppliedTo] = React.useState("");
   210→
   211→  const hasAppliedDateFilter = Boolean(appliedFrom || appliedTo);
   212→  const hasDirtyDateInputs =
   213→    dateFrom !== appliedFrom || dateTo !== appliedTo;
   214→  const canResetDateFilter =
   215→    Boolean(dateFrom || dateTo || appliedFrom || appliedTo || staleOnly);
   216→

   220→  React.useEffect(() => {
   221→    if (typeof window === "undefined") return;

   223→    if (params.get("stale") === "1") {
   224→      setStaleOnly(true);
   225→      params.delete("stale");
   226→      const next = params.toString();
   227→      const nextUrl =
   228→        window.location.pathname + (next ? `?${next}` : "") + window.location.hash;
   229→      window.history.replaceState(null, "", nextUrl);
   230→    }
   231→  }, []);
   232→
   233→  const staleCount = React.useMemo(
   234→    () => leads.filter(isStaleLead).length,
   235→    [leads]
   236→  );
   237→
   238→  const filtered = React.useMemo(() => {
   239→    let list = leads;
   240→    if (staleOnly) {
   241→      list = list.filter(isStaleLead);
   242→    } else if (filter !== "ALL") {
   243→      list = list.filter((l) => l.status === filter);
   244→    }
   245→
   246→    if (appliedFrom || appliedTo) {
   247→      const fromMs = appliedFrom
   248→        ? new Date(appliedFrom + "T00:00:00").getTime()
   249→        : -Infinity;
   250→      const toMs = appliedTo
   251→        ? new Date(appliedTo + "T23:59:59.999").getTime()
   252→        : Infinity;
   253→      list = list.filter((l) => {
   254→        const t = new Date(l.createdAt).getTime();
   255→        return t >= fromMs && t <= toMs;
   256→      });
   257→    }
   258→
   259→    const q = search.trim().toLowerCase();
   260→    if (q) {
   261→      list = list.filter(
   262→        (l) =>
   263→          l.name.toLowerCase().includes(q) ||
   264→          l.phone.toLowerCase().includes(q) ||
   265→          l.email.toLowerCase().includes(q) ||
   266→          (l.businessName ?? "").toLowerCase().includes(q) ||
   267→          (l.serviceName ?? "").toLowerCase().includes(q)
   268→      );
   269→    }
   270→    return list;
   271→  }, [leads, filter, search, appliedFrom, appliedTo, staleOnly]);
   272→

   274→    setAppliedFrom(dateFrom);
   275→    setAppliedTo(dateTo);
   276→  }, [dateFrom, dateTo]);
   277→
   278→  const handleResetDateFilter = React.useCallback(() => {
   279→    setDateFrom("");
   280→    setDateTo("");
   281→    setAppliedFrom("");
   282→    setAppliedTo("");
   283→    setStaleOnly(false);
   284→    setFilter("ALL");
   285→  }, []);
   286→
   287→  // ─── Bulk selection helpers ───────────────────────────────────────────────
   288→  const toggleSelect = React.useCallback((id: string) => {
   289→    setSelectedIds((prev) => {
   290→      const next = new Set(prev);
   291→      if (next.has(id)) next.delete(id);
   292→      else next.add(id);
   293→      return next;
   294→    });
   295→  }, []);
   296→
   297→  const clearSelection = React.useCallback(() => {

   307→    !allVisibleSelected && filtered.some((l) => selectedIds.has(l.id));

   310→    setSelectedIds((prev) => {
   311→      const next = new Set(prev);
   312→      if (filtered.every((l) => next.has(l.id))) {
   313→        for (const l of filtered) next.delete(l.id);
   314→      } else {
   315→        for (const l of filtered) next.add(l.id);
   316→      }
   317→      return next;
   318→    });
   319→  }, [filtered]);
   320→
   321→  const handleRefresh = React.useCallback(() => {
   322→    setSelectedIds(new Set());
   323→    window.location.reload();
   324→  }, []);
   325→
   326→  const selected = React.useMemo(
   327→    () => leads.find((l) => l.id === selectedId) ?? null,
   328→    [leads, selectedId]
   329→  );
   330→
   331→  return (

   336→          <div className="flex flex-wrap gap-1.5">
   337→            {STATUS_FILTERS.map((f) => {
   338→              const active = !staleOnly && filter === f.key;
   339→              const count =
   340→                f.key === "ALL"
   341→                  ? leads.length
   342→                  : leads.filter((l) => l.status === f.key).length;
   343→              return (
   344→                <button
   345→                  key={f.key}
   346→                  onClick={() => {
   347→                    setStaleOnly(false);
   348→                    setFilter(f.key);
   349→                  }}
   350→                  className={cn(
   351→                    "inline-flex items-center gap-1.5 rounded-full px-3 py-1.5 text-xs font-medium transition-colors border",
   352→                    active

   357→                  {f.label}

   375→                  "inline-flex items-center gap-1.5 rounded-full px-3 py-1.5 text-xs font-medium transition-colors border",
   376→                  staleOnly

   380→                title="Lead dengan status Baru/Dihubungi yang belum diperbarui >48 jam"

   386→                    "tabular-nums text-[10px] rounded-full px-1.5 py-0.5",

   584→                            "size-1.5 rounded-full mr-1",
   585→                            LEAD_STATUS[(lead.status as LeadStatusKey) ?? "NEW"]?.dot ??
   586→                              "bg-ink-soft"
   587→                          )}
   588→                        />
   589→                        {LEAD_STATUS[(lead.status as LeadStatusKey) ?? "NEW"]?.label ??
   590→                          lead.status}
   591→                      </Badge>
   592→                    </TableCell>
   593→                    <TableCell className="text-xs text-ink-soft whitespace-nowrap">
   594→                      {relativeDate(lead.createdAt)}
   595→                    </TableCell>
   596→                    <TableCell className="pr-4 text-right">
   597→                      <Button
   598→                        variant="ghost"
   599→                        size="sm"
   600→                        className="h-8"
   601→                        onClick={(e) => {
   602→                          e.stopPropagation();
   603→                          setSelectedId(lead.id);
   604→                        }}
   605→                      >
   606→                        <Eye className="size-4" />
   607→                        <span className="hidden sm:inline">Detail</span>
   608→                      </Button>
   609→                    </TableCell>
   610→                  </TableRow>
   611→                );
   612→              })
   613→            )}
   614→          </TableBody>
   615→        </Table>
   616→      </div>
   617→
   618→      {/* Detail Sheet */}
   619→      <LeadDetailSheet
   620→        lead={selected}
   621→        open={!!selected}
   622→        onOpenChange={(o) => !o && setSelectedId(null)}
   623→        services={services}
   624→      />
   625→
   626→      {/* Floating bulk-action bar */}
   627→      {selectedIds.size > 0 && (
   628→        <LeadsBulkBar
   629→          selectedIds={selectedIds}
   630→          onClear={clearSelection}
   631→          onRefresh={handleRefresh}
   632→        />
   633→      )}
   634→    </div>
   635→  );
   636→}

   638→function LeadDetailSheet({
   639→  lead,
   640→  open,
   641→  onOpenChange,
   642→  services,
   643→}: {
   644→  lead: LeadRow | null;
   645→  open: boolean;
   646→  onOpenChange: (open: boolean) => void;
   647→  services: ServiceOption[];
   648→}) {
   649→  const [status, setStatus] = React.useState<string>(lead?.status ?? "NEW");
   650→  const [note, setNote] = React.useState<string>(lead?.note ?? "");
   651→  const [saving, setSaving] = React.useState(false);
   652→  const [deleting, setDeleting] = React.useState(false);
   653→
   654→  // Sync local state when lead changes
   655→  React.useEffect(() => {
   656→    if (lead) {
   657→      setStatus(lead.status);
   658→      setNote(lead.note);
   659→    }
   660→  }, [lead?.id]);
   661→
   662→  if (!lead) {
   663→    return (
   664→      <Sheet open={open} onOpenChange={onOpenChange}>
   665→        <SheetContent side="right" className="w-full sm:max-w-md" />
   666→      </Sheet>
   667→    );
   668→  }
   669→
   670→  async function handleSave() {
   671→    if (!lead) return;
   672→    setSaving(true);
   673→    try {
   674→      const res = await fetch(`/api/admin/leads/${lead.id}`, {
   675→        method: "PATCH",
   676→        headers: { "Content-Type": "application/json" },
   677→        body: JSON.stringify({ status, note }),
   678→      });
   679→      const data = await res.json().catch(() => ({}));
   680→      if (!res.ok || !data.ok) {
   681→        toast.error(data.error ?? "Gagal menyimpan perubahan.");
   682→        return;
   683→      }
   684→      toast.success("Lead diperbarui.");
   685→      // Refresh to reflect updates on the table
   686→      window.location.reload();
   687→    } catch {
   688→      toast.error("Kesalahan jaringan saat menyimpan.");
   689→    } finally {
   690→      setSaving(false);
   691→    }
   692→  }
   693→
   694→  async function handleDelete() {
   695→    if (!lead) return;
   696→    if (!confirm("Hapus lead ini secara permanen? Tindakan ini tidak bisa dibatalkan.")) {
   697→      return;
   698→    }
   699→    setDeleting(true);
   700→    try {
   701→      const res = await fetch(`/api/admin/leads/${lead.id}`, {
   702→        method: "DELETE",
   703→      });
   704→      const data = await res.json().catch(() => ({}));
   705→      if (!res.ok || !data.ok) {
   706→        toast.error(data.error ?? "Gagal menghapus lead.");
   707→        return;
   708→      }
   709→      toast.success("Lead dihapus.");
   710→      window.location.reload();
   711→    } catch {
   712→      toast.error("Kesalahan jaringan saat menghapus.");
   713→    } finally {
   714→      setDeleting(false);
   715→    }
   716→  }
   717→
   718→  return (
   719→    <Sheet open={open} onOpenChange={onOpenChange}>
   720→      <SheetContent side="right" className="w-full sm:max-w-md flex flex-col p-0">
   721→        <SheetHeader className="px-5 pt-5 pb-3 border-b border-border">
   722→          <SheetTitle className="text-lg text-ink">{lead.name}</SheetTitle>
   723→          <SheetDescription className="sr-only">
   724→            Detail lead dan pembaruan status.
   725→          </SheetDescription>
   726→          <div className="text-xs text-ink-soft flex items-center gap-1.5">
   727→            <Clock className="size-3" />
   728→            {formatDate(lead.createdAt)}
   729→          </div>
   730→        </SheetHeader>
   731→
   732→        {/* Body */}
   733→        <div className="flex-1 overflow-y-auto px-5 py-4 space-y-4">
   734→          {/* Contact info */}
   735→          <div className="space-y-2">
   736→            <h3 className="text-xs font-semibold text-ink-soft uppercase tracking-wider">
   737→              Kontak
   738→            </h3>
   739→            <div className="grid gap-1.5 text-sm">
   740→              <a
   741→                href={`tel:${lead.phone}`}
   742→                className="flex items-center gap-2 text-ink hover:text-navy"
   743→              >
   744→                <Phone className="size-4 text-ink-soft" />
   745→                {lead.phone}
   746→              </a>
   747→              {lead.email && (
   748→                <a
   749→                  href={`mailto:${lead.email}`}
   750→                  className="flex items-center gap-2 text-ink hover:text-navy truncate"
   751→                >
   752→                  <Mail className="size-4 text-ink-soft shrink-0" />
   753→                  <span className="truncate">{lead.email}</span>
   754→                </a>
   755→              )}
   756→              {lead.businessName && (
   757→                <div className="flex items-center gap-2 text-ink">
   758→                  <Building2 className="size-4 text-ink-soft" />
   759→                  {lead.businessName}
   760→                </div>
   761→              )}
   762→            </div>
   763→            {/* WhatsApp click-to-chat with template selector */}
   764→            <WhatsAppTemplateButton
   765→              lead={{
   766→                name: lead.name,
   767→                phone: lead.phone,
   768→                serviceName: lead.serviceName,
   769→                businessName: lead.businessName,
   770→                message: lead.message,
   771→              }}
   772→            />
   773→          </div>
   774→
   775→          {/* Service */}
   776→          {lead.serviceName && (
   777→            <div className="space-y-1.5">
   778→              <h3 className="text-xs font-semibold text-ink-soft uppercase tracking-wider">
   779→                Layanan Diminati
   780→              </h3>
   781→              <Badge
   782→                variant="outline"
   783→                className="bg-navy-50 text-navy border-navy/15"
   784→              >
   785→                {lead.serviceName}
   786→              </Badge>
   787→            </div>
   788→          )}
   789→
   790→          {/* Message */}

   793→              <MessageSquare className="size-3.5" />
   794→              Pesan
   795→            </h3>

   797→              {lead.message}
   798→            </div>
   799→          </div>
   800→
   801→          {/* Status update */}

   803→            <Label className="text-xs font-semibold text-ink-soft uppercase tracking-wider">
   804→              Status
   805→            </Label>

   837→              id="note"
   838→              value={note}
   839→              onChange={(e) => setNote(e.target.value)}
